Pourquoi un StatefulWidget ? L'affichage change quand on clique : le numéro du livre (_index) est un état. Pourquoi le modulo ? (_index + 1) % livres.length vaut 1, 2, 3, 4 puis 0 : retour automatique au premier livre.

Ajouter les images (pubspec.yaml)
  1. Créez un dossier images à la racine du projet (à côté de lib).
  2. Copiez-y vos cinq couvertures (ici des images de remplacement : livre1.png à livre5.png).
  3. Déclarez le dossier dans pubspec.yaml :
flutter:
  uses-material-design: true
  assets:
    - images/

L'indentation compte : deux espaces avant assets:, quatre avant le tiret. Après modification de pubspec.yaml, relancez l'application (un hot reload ne suffit pas).

Points à ne pas rater

  • Chemin d'image faux ou dossier non déclaré dans pubspec.yaml : l'image ne s'affiche pas (erreur « Unable to load asset »).
  • Oublier setState : l'index change mais l'écran reste sur le même livre.
  • Écrire _index++ sans modulo : après le cinquième livre, l'application plante (index hors limites).
  • Mettre padding sur un Card : ce paramètre n'existe pas, utilisez un widget Padding.
  • ClipOval sans width égal à height donne une ellipse, pas un cercle.

TP3, exercice 1 : calculateur d'IMC

Formule : IMC = poids (kg) / taille (m) au carré. La taille est saisie en centimètres : il faut la diviser par 100.

Solution

Code complet (imc_app.dart)
import 'package:flutter/material.dart';

// TP3, exercice 1 : calculateur d'IMC
// Lancer : flutter run -t lib/main_imc.dart

/// IMC = poids (kg) / taille (m) au carré. La taille est saisie en centimètres.
double calculerImc(double poids, double tailleCm) {
  final double tailleM = tailleCm / 100;
  return poids / (tailleM * tailleM);
}

/// Message selon le tableau du TP, par exemple : "Poids idéal (IMC=23.0)".
String getMessage(double imc) {
  String categorie;
  if (imc < 18.6) {
    categorie = 'Poids très bas';
  } else if (imc < 24.9) {
    categorie = 'Poids idéal';
  } else if (imc < 29.9) {
    categorie = 'Surpoids';
  } else if (imc < 34.9) {
    categorie = 'Obésité grade 1';
  } else if (imc < 39.9) {
    categorie = 'Obésité grade 2';
  } else {
    categorie = 'Obésité grade 3';
  }
  return '$categorie (IMC=${imc.toStringAsFixed(1)})';
}

class ImcApp extends StatelessWidget {
  const ImcApp({super.key});

  @override
  Widget build(BuildContext context) {
    return MaterialApp(
      debugShowCheckedModeBanner: false,
      title: 'IMC Calculator',
      theme: ThemeData(colorSchemeSeed: Colors.deepOrange),
      home: const EcranImc(),
    );
  }
}

class EcranImc extends StatefulWidget {
  const EcranImc({super.key});

  @override
  State<EcranImc> createState() => _EcranImcState();
}

class _EcranImcState extends State<EcranImc> {
  final _formKey = GlobalKey<FormState>(); // pour appeler validate()
  final _poidsController = TextEditingController();
  final _tailleController = TextEditingController();
  String _message = '';

  String? _valider(String? valeur) {
    if (valeur == null || valeur.isEmpty) {
      return 'Prière de saisir une valeur';
    }
    final double? nombre = double.tryParse(valeur);
    if (nombre == null || nombre <= 0) {
      return 'Valeur invalide';
    }
    return null; // null = valeur correcte
  }

  void _calculer() {
    if (_formKey.currentState!.validate()) {
      final double poids = double.parse(_poidsController.text);
      final double taille = double.parse(_tailleController.text);
      setState(() {
        _message = getMessage(calculerImc(poids, taille));
      });
    } else {
      setState(() {
        _message = '';
      });
    }
  }

  @override
  void dispose() {
    _poidsController.dispose();
    _tailleController.dispose();
    super.dispose();
  }

  @override
  Widget build(BuildContext context) {
    return Scaffold(
      backgroundColor: Colors.deepOrange,
      body: SafeArea(
        child: SingleChildScrollView(
          child: Padding(
            padding: const EdgeInsets.all(24),
            child: Column(
              crossAxisAlignment: CrossAxisAlignment.start,
              children: [
                const SizedBox(height: 16),
                const Text(
                  'IMC CALCULATOR',
                  style: TextStyle(
                    color: Colors.white,
                    fontSize: 28,
                    fontWeight: FontWeight.bold,
                  ),
                ),
                const Text(
                  'Contrôler votre poids',
                  style: TextStyle(color: Colors.white70, fontSize: 16),
                ),
                const SizedBox(height: 32),
                Card(
                  shape: RoundedRectangleBorder(
                    borderRadius: BorderRadius.circular(20),
                  ),
                  child: Padding(
                    padding: const EdgeInsets.all(20),
                    child: Form(
                      key: _formKey,
                      child: Column(
                        children: [
                          TextFormField(
                            controller: _poidsController,
                            keyboardType: TextInputType.number,
                            decoration: const InputDecoration(
                              labelText: 'Poids en Kg',
                            ),
                            validator: _valider,
                          ),
                          const SizedBox(height: 12),
                          TextFormField(
                            controller: _tailleController,
                            keyboardType: TextInputType.number,
                            decoration: const InputDecoration(
                              labelText: 'Taille en cm',
                            ),
                            validator: _valider,
                          ),
                          const SizedBox(height: 24),
                          ElevatedButton(
                            onPressed: _calculer,
                            style: ElevatedButton.styleFrom(
                              backgroundColor: Colors.deepOrange,
                              foregroundColor: Colors.white,
                              shape: const StadiumBorder(),
                              padding: const EdgeInsets.symmetric(
                                horizontal: 28,
                                vertical: 14,
                              ),
                            ),
                            child: const Text('Calculer votre IMC'),
                          ),
                          const SizedBox(height: 16),
                          Text(
                            _message,
                            textAlign: TextAlign.center,
                            style: const TextStyle(
                              fontSize: 18,
                              fontWeight: FontWeight.bold,
                            ),
                          ),
                        ],
                      ),
                    ),
                  ),
                ),
              ],
            ),
          ),
        ),
      ),
    );
  }
}
  1. Interface : deux TextFormField dans un Form, un ElevatedButton et un Text pour le message.
  2. calculerImc : convertit la taille en mètres puis divise.
  3. getMessage : une chaîne if / else if du plus petit seuil au plus grand, puis le format « catégorie (IMC=valeur) ».
  4. Bouton : validate(), puis setState pour afficher le message.
  5. Validation : un validator qui refuse une valeur vide, non numérique ou nulle.

Incohérences du sujet et pièges

  • L'exemple du sujet dit « Poids normal (IMC=23) » mais le tableau dit « Poids idéal ». J'ai suivi le tableau.
  • Les bornes sont des inférieurs stricts : 18.6 est « Poids idéal », 24.9 est « Surpoids » (testé).
  • Le tableau laisse un trou entre 39.9 et 40 (par exemple 39.95). Mon else final classe ce cas en grade 3.
  • Oublier de diviser la taille par 100 donne un IMC de 0.007 : toujours « Poids très bas ».
  • Un champ vide fait planter double.parse : d'où la validation avant le calcul. double.tryParse renvoie null au lieu de planter.

// Série 1 : exercices Dart en console. Lancer : dart run lib/dart_series.dart
import 'dart:math';

// Calculer F(x) = 5x + 8
double f(double x) => 5 * x + 8;

// Mention correspondant à une note (seuils choisis : le cours ne donne que les noms)
String mention(double note) {
  if (note < 10) {
    return 'Ajourné';
  } else if (note < 12) {
    return 'Passable';
  } else if (note < 14) {
    return 'Assez bien';
  } else if (note < 16) {
    return 'Bien';
  } else if (note < 18) {
    return 'Très bien';
  } else {
    return 'Excellent';
  }
}

// Gestion des formes : classe abstraite, interface, classes dérivées
abstract class Shape {
  String color;
  Shape(this.color);

  double calculateArea();
  void displayInfo();
}

abstract interface class Drawable {
  void draw();
}

class Circle extends Shape implements Drawable {
  double radius;
  Circle(super.color, this.radius);

  @override
  double calculateArea() => pi * radius * radius;

  @override
  void displayInfo() {
    print('Cercle $color, aire = ${calculateArea().toStringAsFixed(2)}');
  }

  @override
  void draw() {
    print('Je dessine un cercle $color');
  }
}

class Rectangle extends Shape implements Drawable {
  double largeur;
  double hauteur;
  Rectangle(super.color, this.largeur, this.hauteur);

  @override
  double calculateArea() => largeur * hauteur;

  @override
  void displayInfo() {
    print('Rectangle $color, aire = ${calculateArea().toStringAsFixed(2)}');
  }

  @override
  void draw() {
    print('Je dessine un rectangle $color');
  }
}

void main() {
  // F(x)
  print('F(2) = ${f(2)}');

  // Mentions
  for (final note in [8.0, 11.0, 13.5, 15.0, 17.0, 19.0]) {
    print('$note -> ${mention(note)}');
  }

  // Dictionnaire de contacts
  final Map<String, String> contacts = {
    'Ahmed': '0652417885',
    'Fatine': '0654872183',
    'Louiza': '0647852135',
  };
  contacts['Samira'] = '0685236987'; // ajouter
  contacts['Ahmed'] = '0600000000'; // modifier
  contacts.remove('Fatine'); // supprimer
  if (contacts.containsKey('Samira')) {
    print('Samira : ${contacts['Samira']}');
  }
  for (final entree in contacts.entries) {
    print('${entree.key} : ${entree.value}');
  }

  // Formes
  final List<Shape> formes = [Circle('rouge', 2), Rectangle('bleu', 3, 4)];
  for (final forme in formes) {
    forme.displayInfo();
    if (forme is Drawable) {
      (forme as Drawable).draw();
    }
  }
}
