Exercice 2 : mini-algorithmes

Somme des multiples, factorielles, chiffres d'un nombre, nombre premier.

Solution détaillée

algos.php

<?php
// 1. somme des multiples de 3 ou 5 inférieurs à 1000
$somme = 0;
for ($i = 1; $i < 1000; $i++) {
    if ($i % 3 === 0 || $i % 5 === 0) { $somme += $i; }
}
echo "Somme : $somme\n";                                   // 233168

// 2. factorielles de 0 à 12, puis plus petit n tel que n! > 1 000 000
$f = 1;
for ($n = 0; $n <= 12; $n++) {
    if ($n > 0) { $f *= $n; }
    echo "$n! = $f\n";
}
$n = 0; $f = 1;
while (true) {
    $n++; $f *= $n;
    if ($f > 1000000) { break; }
}
echo "Plus petit n : $n ($n! = $f)\n";                      // 10 (3628800)

// 3. chiffres d'un nombre (do ... while : un nombre a au moins un chiffre, même 0)
foreach ([12345, 0] as $nombre) {
    $reste = $nombre; $nbChiffres = 0; $somChiffres = 0;
    do {
        $somChiffres += $reste % 10;
        $reste = intdiv($reste, 10);
        $nbChiffres++;
    } while ($reste > 0);
    echo "$nombre : $nbChiffres chiffre(s), somme = $somChiffres\n";  // 5 et 15 ; puis 1 et 0
}

// 4. nombre premier
foreach ([97, 91] as $p) {
    $premier = $p >= 2; $diviseur = null;
    for ($d = 2; $d < $p; $d++) {
        if ($p % $d === 0) { $premier = false; $diviseur = $d; break; }
    }
    echo $premier ? "$p est premier.\n" : "$p n'est pas premier (divisible par $diviseur).\n";
}

Valeurs de contrôle retrouvées

Somme des multiples de 3 ou 5 : 233168 | plus petit n tel que n! dépasse un million : 10 (3628800) | 12345 : 5 chiffres, somme 15 | 97 est premier | 91 n'est pas premier (divisible par 7)

Pourquoi do ... while pour les chiffres ? Tout nombre a au moins un chiffre, même 0 : le bloc doit s'exécuter une fois avant de tester. Avec $nombre = 0, on obtient 1 chiffre et une somme de 0 (un while donnerait 0 chiffre).

<?php

    ["nom" => "Rim Alaoui",    "filiere" => "GE", "notes" => ["PHP" => 19, "SQL" => 13, "Réseaux" => 15]],
];

// 1. moyenne pondérée
function moyennePonderee(array $notes, array $coefs): float
{
    $somme = 0; $total = 0;
    foreach ($notes as $module => $note) {
        $somme += $note * $coefs[$module];
        $total += $coefs[$module];
    }
    return $total === 0 ? 0 : $somme / $total;
}

// 2. mention
function mention(float $moyenne): string
{
    return match (true) {
        $moyenne >= 16 => "Très bien",
        $moyenne >= 14 => "Bien",
        $moyenne >= 12 => "Assez bien",
        $moyenne >= 10 => "Passable",
        default        => "Non validé",
    };
}

// 3. ajouter la moyenne : on modifie $promo[$i], pas la copie $e
foreach ($promo as $i => $e) {
    $promo[$i]["moyenne"] = round(moyennePonderee($e["notes"], $coefs), 2);
}

// 4. tri décroissant par moyenne, puis tableau
usort($promo, fn($a, $b) => $b["moyenne"] <=> $a["moyenne"]);

echo "<table border='1'><tr><th>Rang</th><th>Nom</th><th>Filière</th>";
foreach ($coefs as $module => $c) { echo "<th>$module</th>"; }
echo "<th>Moyenne</th><th>Mention</th></tr>";
foreach ($promo as $i => $e) {
    echo "<tr><td>" . ($i + 1) . "</td><td>{$e['nom']}</td><td>{$e['filiere']}</td>";
    foreach ($coefs as $module => $c) { echo "<td>{$e['notes'][$module]}</td>"; }
    echo "<td>{$e['moyenne']}</td><td>" . mention($e['moyenne']) . "</td></tr>";
}
echo "</table>";

// 5. major de chaque filière : la promo est triée, le premier rencontré est le major
$majors = [];
foreach ($promo as $e) {
    if (!isset($majors[$e["filiere"]])) { $majors[$e["filiere"]] = $e["nom"]; }
}

// 6. moyenne d'un module avec array_map
function moyenneModule(array $promo, string $module): float
{
    $notes = array_map(fn($e) => $e["notes"][$module], $promo);
    return array_sum($notes) / count($notes);
}

echo "Majors : ";
foreach ($majors as $f => $nom) { echo "$f : $nom | "; }
echo "\nMoyennes par module : ";

echo "\n";

    "CLV" => ["nom" => "Clavier",    "prix" => 150,  "qte" => 10],
    "SOU" => ["nom" => "Souris",     "prix" => 80,   "qte" => 8],
    "ECR" => ["nom" => "Écran",      "prix" => 1200, "qte" => 2],
    "CAB" => ["nom" => "Câble HDMI", "prix" => 45,   "qte" => 0],
];

$panier = [];

// 1. & : la fonction doit MODIFIER le panier et le stock de l'appelant, pas des copies
function ajouterAuPanier(array &$panier, array &$stock, string $ref, int $qte): string
{
    if (!isset($stock[$ref])) {
        return "Référence inconnue : $ref";
    }
    $nom = $stock[$ref]["nom"];
    if ($qte <= 0) {
        return "Quantité invalide pour $nom";
    }
    if ($qte > $stock[$ref]["qte"]) {
        return "Stock insuffisant pour $nom ({$stock[$ref]['qte']} disponible(s))";
    }
    $stock[$ref]["qte"] -= $qte;
    $panier[$ref] = ($panier[$ref] ?? 0) + $qte;      // cumul si déjà présent
    return "OK : $qte x $nom";
}

// 3. traitement des commandes
foreach ($commandes as [$ref, $qte]) {
    $msg = ajouterAuPanier($panier, $stock, $ref, $qte);
    $couleur = str_starts_with($msg, "OK") ? "green" : "red";
    echo "<p style='color:$couleur'>$msg</p>";
}

// 4. total avec array_reduce sur les clés du panier
function totalPanier(array $panier, array $stock): float
{
    return array_reduce(array_keys($panier),
        fn($acc, $ref) => $acc + $panier[$ref] * $stock[$ref]["prix"], 0);
}

$sousTotal = totalPanier($panier, $stock);
// 5. remise

echo "Panier : ";
foreach ($panier as $ref => $q) { echo "$q {$stock[$ref]['nom']}, "; }
echo "| " . array_sum($panier) . " articles\n";
echo "Sous-total : $sousTotal DH | Remise ($taux %) : -$remise DH | Total : " . ($sousTotal - $remise) . " DH\n";

// 6. produits en rupture

    fclose($f);

        $erreur = "Aucun fichier reçu.";
